# What a hand-off tree holds, besides the specs and the stubs (make-handoff.sh).
# Paths are git pathspecs relative to the repository root, taken from HEAD.
#   include P        archive P
#   exclude P        leave P out of the included paths
#   exclude-glob P   leave out paths matching the glob P (* stays in one directory)
#   keep P           put P back although an exclude matched it
#   placeholder P    write an empty file at P (a kept test names it at compile
#                    time; the real file is MuPDF output and stays out)
#
# Tests and the data they read: make-handoff.sh keeps only the requested
# workstreams' acceptance set, read from docs/clean-room/acceptance/
# {01-03,04-08}.md (test files, support modules, tests/data, fixtures and
# the engine/reference files listed there); acceptance-files.py parses it.
# Everything else under engine/reference/ stays out.

include engine
include fixtures
include assets/fonts/DroidSansFallback.ttf

# Oracles, design notes and tools (MuPDF-informed or AGPL-only).
exclude engine/oracle
exclude engine/design
exclude engine/tools
exclude engine/NOTICE
exclude engine/COPYING
# Design and review documents; engine/README.md is replaced by a neutral one.
exclude-glob engine/*.md
exclude engine/target
exclude engine/bridge/pkg

# Reference data: MuPDF evidence, oracle output and old reports.
exclude engine/reference
# Third-party licence texts (not MuPDF output).
keep engine/reference/licenses
# These two notices name the MuPDF source tarball the URW data came from.
exclude engine/reference/licenses/urw-metrics/NOTICE
exclude engine/reference/licenses/urw-outlines/source-notices.txt

# Fixtures: the documents only. Generators and provenance notes name the
# tools (some MuPDF) that wrote them.
exclude-glob fixtures/**/*.py
exclude-glob fixtures/**/*.mjs
exclude-glob fixtures/**/*.sh
exclude-glob fixtures/**/*.provenance.json
exclude-glob fixtures/**/README.md
exclude-glob fixtures/*.py
exclude-glob fixtures/*.mjs
exclude-glob fixtures/*.sh
exclude-glob fixtures/*.provenance.json
exclude fixtures/p3/manifest.json
